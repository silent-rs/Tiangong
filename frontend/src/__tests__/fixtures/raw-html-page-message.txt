    // 接口统一返回 { code, message, data, traceId }，HTTP 状态码始终有意义。
    async function call(method, path, body) {
      const res = await fetch(path, {
        method,
        headers: body ? { "content-type": "application/json" } : {},
        body: body ? JSON.stringify(body) : undefined,
      });
      const json = await res.json();
      if (!res.ok || json.code >= 400) {
        const detail = json.data && json.data.title ? "：" + json.data.title.map(function (e) { return e.message; }).join("；") : "";
        throw new Error(json.message + detail);
      }
      return json.data;
    }

    async function refresh() {
      const articles = await call("GET", "/api/articles");
      document.getElementById("articles").innerHTML = articles.items.map(function (a) {
        return "<li><strong></strong><div class=\"meta\"></div></li>";
      }).join("");
      const nodes = document.getElementById("articles").children;
      articles.items.forEach(function (a, i) {
        nodes[i].querySelector("strong").textContent = a.title;
        nodes[i].querySelector(".meta").textContent = a.status === 1 ? "已发布" : "草稿";
      });
    }

    document.getElementById("create").addEventListener("submit", async function (event) {
      event.preventDefault();
      const title = document.getElementById("title");
      document.getElementById("error").textContent = "";
      try {
        await call("POST", "/api/articles", { title: title.value });
        title.value = "";
        await refresh();
      } catch (e) {
        document.getElementById("error").textContent = e.message;
      }
    });

    refresh().catch(function (e) {
      document.getElementById("error").textContent = e.message + "（先执行一次 `cargo run -- migrate up`）";
    });
  </script>
